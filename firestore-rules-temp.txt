rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Definimos funciones para verificar roles
    function isSuperAdmin() {
      return request.auth != null && request.auth.token.email in ['elinquisidor09@gmail.com', 'milkavv.2001@gmail.com'];
    }
    
    function isSuperVendedor() {
      return request.auth != null && request.auth.token.email in ['garyvv.1993@gmail.com', 'abrahamjimenez092003@gmail.com'];
    }
    
    function isVendedor() {
      return request.auth != null && request.auth.token.email in ['amvf280194@gmail.com', 'noeatlas28@gmail.com'];
    }
    
    // Función combinada para verificar si es admin (cualquier tipo)
    function isAdmin() {
      return isSuperAdmin() || isSuperVendedor();
    }
    
    // Función para verificar acceso a ventas
    function hasAccessToSales() {
      return isAdmin() || isVendedor();
    }

    // Colección de Productos
    match /products/{productId} {
      allow read: if request.auth != null;
      // Super admins pueden escribir productos completamente
      allow write: if isSuperAdmin();
      // Super vendedores y vendedores pueden actualizar solo el stock (para ventas)
      allow update: if hasAccessToSales() && 
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock']) &&
        request.resource.data.stock >= 0;
    }

    // Colección de Ventas
    match /sales/{saleId} {
      // Todos los tipos de admin y vendedores pueden leer y crear ventas
      allow read: if hasAccessToSales();
      allow create: if hasAccessToSales();
      // Admins pueden modificar/eliminar ventas
      allow update, delete: if isAdmin();
    }

    // NUEVA REGLA PARA LA COLECCIÓN DE CATEGORÍAS
    match /categories/{categoryId} {
      // Cualquiera puede leer las categorías para mostrarlas en la tienda
      allow read: if true;
      // Solo super admins pueden gestionar categorías
      allow write: if isSuperAdmin();
    }

    // Colección de Egresos (Solo para Super Admin y Super Vendedor)
    match /expenses/{expenseId} {
      // Solo admins (Super Admin y Super Vendedor) pueden leer, escribir y eliminar egresos
      allow read, write, delete: if isAdmin();
    }
  }
}
